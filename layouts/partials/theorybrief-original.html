{{- /* The Theory Brief original of a mirrored blog post, or "". It's the `original` front-matter URL if set (`original: false` turns it off; an empty value, as the admin saves it, counts as unset), else the Theory Brief post with the same title (lowercased, non-alphanumerics stripped) in static/feeds/theorybrief.json. Used for the "also appears in" line, the canonical URL, and leaving such posts out of the sitemap. Context: the page. */ -}} {{- $original := "" -}} {{- if and .IsPage (eq .Section "blog") -}} {{- $param := printf "%v" .Params.original -}} {{- if eq $param "false" -}} {{- else if .Params.original -}} {{- $original = .Params.original -}} {{- else -}} {{- $key := lower .Title | replaceRE `[^a-z0-9]` "" -}} {{- range (transform.Unmarshal (readFile "static/feeds/theorybrief.json")).posts -}} {{- if eq (lower .title | replaceRE `[^a-z0-9]` "") $key }}{{ $original = .link }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $original -}}
